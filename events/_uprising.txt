# uprising
    country_event = {
        id = uprising.1
        title = "Uprising in [uprising_country.GetNameDef]!"
        desc = uprising.1.d

        is_triggered_only = yes
        trigger = { is_subject = no }

        option = {
            name = "To arms!"
            add_war_support = 0.05
            declare_war_on = { target = event_target:uprising_country type = take_claimed_state }
        }
        option = {
            name = "Let them do as they wish."
            add_political_power = -15
            
            every_state = {
                limit = { is_owned_by = event_target:uprising_country is_core_of = ROOT }
                remove_core_of = ROOT
            }
            every_state = {
                limit = { is_owned_by = event_target:uprising_country is_claimed_by = ROOT }
                remove_claim_by = ROOT
            }

            # ai
            ai_chance = { modifier = { not = { has_government = democratic } factor = 0 } }
        }
    }
