# soviet
    news_event = {
        id = rus_soviet.1
        title = "Communists Seize the Tula Armoury!"

        is_triggered_only = yes
        fire_only_once = yes
        major = yes

        immediate = {
            RUS = { 
                # other soviets
                news_event = { id = ukr_soviet.1 days = 12 } 
                news_event = { id = blr_uprising.1 days = 26 }

                # alash
                news_event = { id = kaz_uprising.1 months = 1 random_days = 14 }
            }
            SOD = {
                country_event = { id = rus_coalition.1 months = 6 }
            }
            SOV = {
                set_capital = { state = 223 }

                # states
                every_state = {
                    limit = { or = { state = 40 state = 205 state = 223 state = 224 state = 242 state = 254 state = 402 state = 561 state = 568 state = 589 state = 615 } }
                    set_variable = { original_controller = controller }     transfer_state_to = PREV
                }

                # war
                every_country = { 
                    limit = { original_tag = RUS }
                    declare_war_on = { target = PREV type = civil_war } 
                }

                # other
                division_template = { name = "Worker Militias" is_locked = yes regiments = { infantry = { x = 0 y = 0 } infantry = { x = 0 y = 1 } infantry = { x = 1 y = 0 } infantry = { x = 1 y = 1 } } }
                every_owned_state = { 
                    create_unit = { division = "name = \"Worker Militias\" division_template = \"Worker Militias\" start_experience_factor = 0.2" count = 4 owner = PREV } 
                    var:original_controller = { set_state_controller = PREV }
                }

                create_faction = "Communist Internationale"
            }

            # troops
            RUS = {
                transfer_units_fraction = {
                    target = SOV
                    size = 0.25      navy_ratio = 0.05      stockpile_ratio = 0.1
                }
            }
        }

        option = {
            name = "Ok."
        }
    }
