# Demilitarization of Albania
    add_namespace = demilitarization_of_albania

    # starting event
        country_event = {
            id = demilitarization_of_albania.1
            title = "Demilitarization of Albania"
            desc = demilitarization_of_albania.1.d

            trigger = {
                tag = SER
                44 = { 
                    is_claimed_by = SER 
                    owner = { 
                        has_idea = idea_war_loser 
                        controls_state = 44
                    }
                }
            }
            fire_only_once = yes

            option = {
                name = "Demand withdrawal of the troops."
                every_country = {
                    limit = { owns_state = 44 }
                    country_event = { id = demilitarization_of_albania.2 }
                }
            }
            option = {
                name = "This is unnecessary."
                add_war_support = 0.05
            }
        }

    # demilitarization option
        country_event = {
            id = demilitarization_of_albania.2
            title = "Serbia Demands Troops Out of Albania"
            desc = demilitarization_of_albania.2.d

            is_triggered_only = yes

            option = {
                name = "Let them have it."
                add_war_support = -0.025
                44 = { set_demilitarized_zone = yes }
                
                hidden_effect = { news_event = { id = demilitarization_of_albania.3 hours = 6 } }
            }
            option = {
                name = "And what else?"
                FROM = { create_wargoal = { target = ROOT type = take_claimed_state generator = { 44 } } }
            }
        }

        news_event = {
            id = demilitarization_of_albania.3
            title = "[TUR.GetAdjective] Troops Out of Albania"
            desc = demilitarization_of_albania.3.d

            is_triggered_only = yes
            major = yes

            option = {
                name = "Interesting."
            }
        }
