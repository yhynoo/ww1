        id = armistice_demands.1
        title = "Opposition Demands an Armistice"

        is_triggered_only = yes
        immediate = { hidden_effect = { add_timed_idea = { idea = idea_armistice_timeout months = 2 } } }

        option = {
            name = "We must agree."

    }
