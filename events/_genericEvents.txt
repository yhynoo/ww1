            id = generic.9
            title = "Time for Armistice?"
            desc = generic.9.d

            is_triggered_only = yes
            trigger = { can_have_armistice_event = yes }
            immediate = { 
                hidden_effect = { add_timed_idea = { idea = idea_armistice_timeout months = 6 } } 
            }

            option = {
                name = "This war has lasted too long."
                every_enemy_country = {
                    limit = { or = { is_faction_leader = yes is_in_faction = no is_subject = no } }
                    country_event = { id = diplomacy.8 hours = 12 }
                }
            }
            option = {
                name = "We must continue."
                if = {
                    limit = { not = { has_idea = idea_war_exhaustion } }
                    add_ideas = idea_war_exhaustion
                }
                else = {
                    add_popularity = { ideology = ROOT popularity = -0.05 }
                }
            }
        }
