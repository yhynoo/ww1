# armistice
    country_event = {
        id = generic.9
        title = "The Offensive Effort Stalls"
        desc = generic.9.d

        trigger = {
            longest_war_length > 5          # already after six months
            has_offensive_war = yes

            is_subject = no
            any_enemy_country = {
                is_armistice_target = yes
                not = { surrender_progress > 0.05 }
            }
            not = { has_idea = idea_armistice_timeout }
        }
        immediate = { 
            hidden_effect = { add_timed_idea = { idea = idea_armistice_timeout days = 180 } } 
        }

        option = {
            name = "We shall ask for an armistice."
            every_enemy_country = {
                limit = { is_armistice_target = yes }
                
                custom_effect_tooltip = tt_ask_for_armistice
                hidden_effect = { country_event = { id = armistice.1 days = 2 } }
            }

            # ai
            ai_chance = { modifier = { has_war_support > 0.25  factor = 0 } }
        }
        option = {
            name = "We must fight until we achieve our goals."
            add_political_power = -50
        }
    }

    country_event = {
        id = generic.10
        title = "Opposition Demands an Armistice"
        desc = generic.10.d

        trigger = {
            is_subject = no
            or = {
                and = { has_war = yes has_war_support < 0.4 longest_war_length > 11 }
                surrender_progress > 0.15
            }
            any_enemy_country = { is_armistice_target = yes }

            check_variable = { highest_party_popularity@exclude_ruling_party > 0.25 }
            not = { has_idea = idea_armistice_timeout }
        }
        immediate = { 
            hidden_effect = { add_timed_idea = { idea = idea_armistice_timeout days = 180 } } 
        }

        option = {
            name = "They are right."
            every_enemy_country = {
                limit = { is_armistice_target = yes }
                
                custom_effect_tooltip = tt_ask_for_armistice
                country_event = { id = armistice.1 days = 2 }
            }
        }
        option = {
            name = "The war is not over yet."
            add_popularity = { ideology = ROOT popularity = -0.05 }
        }
    }
