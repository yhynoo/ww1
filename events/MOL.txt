        id = mol_uprising.1
        title = "Uprising in Moldova!"

        is_triggered_only = yes
        major = yes

        immediate = {
            MOL = {
                # states
                every_core_state = {
                    set_variable = { original_controller = controller }     transfer_state_to = PREV
                }

                # war
                every_country = {
                    limit = { original_tag = RUS }
                    declare_war_on = { target = PREV type = annex_everything }

                load_oob = MOL

                every_owned_state = { 
                    var:original_controller = { set_state_controller = PREV }
