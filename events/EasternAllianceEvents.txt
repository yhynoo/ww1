# Eastern Alliance
add_namespace = east_alliance

    # starting event
        country_event = {
            id = east_alliance.1
            title = "The Defense Against [AUS.GetNameDef]"
            desc = east_alliance.1.d

            trigger = {
                tag = SER
                is_in_faction = no
                AUS = { focus_progress = { focus = AUS_take_ours progress > 0.4 } }
            }
            fire_only_once = yes

            option = {
                trigger = { RUS = { is_in_faction = no } }
                name = "Appeal to [RUS.GetNameDef]"
                RUS = { 
                    country_event = { id = east_alliance.2 hours = 6 } 
                }
            }
            option = {
                trigger = { FRA = { has_opinion = { target = SER value > 20 } is_faction_leader = yes } }
                name = "Seek to join the Entente."
                FRA = {

                }
            }
            option = {
                name = "We will manage ourselves."
                add_war_support = 0.05
            }
        }

    # appeal to the tsar
        country_event = {
            id = east_alliance.2
            title = "The Serbian Appeal"
            desc = east_alliance.2.d

            is_triggered_only = yes

            option = {
                name = "Inititate alliance talks."
                create_faction = "The Eastern Agreement"

                every_country = {
                    limit = { or = { tag = GRE tag = ROM tag = SER } is_subject = no is_in_faction = no }
                    country_event = { id = east_alliance.3 hours = 6 }
                }

                hidden_effect = {
                    news_event = { id = east_alliance.5 hours = 12 }
                }
            }
            option = {
                name = "This is too much."
                SER = { country_event = { id = east_alliance.5 hours = 12 } }
            }
        }

        country_event = {
            id = east_alliance.3
            title = "Invitation to the Eastern Agreement"
            desc = east_alliance.3.d

            is_triggered_only = yes

            option = {
                name = "Good idea."
                add_war_support = 0.05
                FROM = { add_to_faction = ROOT }
            }
            option = {
                name = "We are not interested."
                FROM = { country_event = { id = east_alliance.6 hours = 6 } }

                # Serbia never declines
                ai_chance = { modifier = { tag = SER } factor = 0 }
            }
        }

    # entente
        country_event = {
            id = east_alliance.4
            title = "[SER.GetName] Seeks Access to the Entente"
            desc = east_alliance.4.d

            is_triggered_only = yes

            option = {
                name = "Alright."
                add_to_faction = FROM
            }
            option = {
                name = "This is too dangerous."
                SER = { county_event = { id = east_alliance.6 hours = 6 } }
            }
        }

    # announcement
        news_event = {
            id = east_alliance.5
            title = "The Eastern Agreement Established"
            desc = east_alliance.5.d

            is_triggered_only = yes
            major = yes

            option = {
                name = "Interesting."
            }
        }

    # declined
        country_event = {
            id = east_alliance.6
            title = "[FROM.GetName] Declines Our Offer"
            desc = east_alliance.6.d

            is_triggered_only = yes

            option = {
                name = "Here goes nothing."
                add_political_power = -15
            }
        }
