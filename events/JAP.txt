    # Twenty-One Demands
        country_event = {
            id = japan.7
            title = "The Army Staff's Demand"
            desc = japan.7.d

            is_triggered_only = yes

            option = {
                name = "Demand their submission."
                custom_effect_tooltip = tt_twenty_one_demands_maximalist

                hidden_effect = {
                    set_country_flag = maximalist_demands
                    MAN = { country_event = { id = japan.8 hours = 6 } }
                }
            }
            option = {
                name = "Ask for limited concessions."
                custom_effect_tooltip = tt_twenty_one_demands_small

                hidden_effect = {
                    MAN = { country_event = { id = japan.8 hours = 6 } }
                }
            }
        }

        country_event = {
            id = japan.8
            title = "A Delegation from Tokyo"
            desc = japan.8.d

            is_triggered_only = yes

            option = {
                name = "We must agree to this."

                if = {
                    limit = { JAP = { has_country_flag = maximalist_demands } }
                    JAP = { set_autonomy = { target = MAN autonomous_state = autonomy_dominion } }
                    every_state = {
                        limit = { 
                            or = { is_controlled_by = ROOT controller = { is_subject_of = ROOT } }
                            or = { state = 296 state = 619 state = 328 state = 610 } 
                        }
                        transfer_state_to = JAP
                    }
                }
                else = {
                    every_state = {
                        limit = { 
                            or = { is_controlled_by = ROOT controller = { is_subject_of = ROOT } }
                            or = { state = 592 state = 593 state = 594 state = 605 } 
                        }
                        ROOT = { give_resource_rights = { receiver = JAP state = PREV } }
                    }
                }

                # atmosphere event
                hidden_effect = { FROM = { news_event = { id = japan.9 hours = 6 } } }
            }
            option = {
                name = "Send them away."
                JAP = { create_wargoal = { target = ROOT type = puppet_wargoal_focus } }
            }
        }
