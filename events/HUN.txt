    news_event = {

        major = yes

        }
    }

# communism
    news_event = {
        id = hun_communism.1
        title = "Communist Uprising in Hungary!"

        is_triggered_only = yes
        major = yes

        immediate = {
            HUN = {
                start_civil_war = { 
                    ideology = communism states = { 43 70 } army_ratio = 0.4

                    # extra stuff
                    division_template = { name = "Worker Militias" is_locked = yes regiments = { infantry = { x = 0 y = 0 } infantry = { x = 0 y = 1 } } }
                    every_owned_state = { create_unit = { division = "name = \"Worker Militias\" division_template = \"Worker Militias\" start_experience_factor = 0.2" count = 1 owner = PREV } }

                    # control
                    every_owned_state = { ROOT = { set_state_controller = PREV } }
                }
            }
        }

        option = {
            name = "Ok."
        }
    }

# independence
    news_event = {
        id = hun_independence.1
        title = "End of the Dual Monarchy"

        is_triggered_only = yes
        major = yes

        immediate = {
            every_state = {
                limit = { 
                    or = { is_core_of = HUN state = 109 }
                    is_owned_by = AUS 
                }

                HUN = { set_state_owner = PREV }    remove_core_of = AUS
                if = { limit = { is_fully_controlled_by = AUS } HUN = { set_state_controller = PREV } }
            }

            AUS = {
                transfer_units_fraction = { target = HUN    size = 0.4    stockpile_ratio = 0.25 }
                faction_leader = { add_to_faction = HUN }

                drop_cosmetic_tag = yes
            }

            HUN = {
                news_event = { id = hun_armistice.1 days = 14 }

                # uprisings
                news_event = { id = pol_galicia.1 days = 6 }
                news_event = { id = cro_uprising.1 days = 11 }
            }
        }

        option = {
            name = "Ok."
        }
    }
