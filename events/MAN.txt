## WARLORDISM EVENTS
    # [warlord state] Declares Autonomy
        country_event = {
            id = manchukuo.18
            title = "[FROM.GetName] Declares Autonomy"
            desc = manchukuo.18.d

            is_triggered_only = yes
            immediate = {
                if = {
                    limit = { not = { has_country_flag = in_warlord_era } }
                    set_country_flag = in_warlord_era
                }
            }

            option = {
                name = "This is a satisfactory solution for now."
                add_stability = 0.05
                set_autonomy = { target = FROM autonomous_state = autonomy_dominion }

                hidden_effect = {
                    FROM = { 
                        set_politics = { ruling_party = monarchism elections_allowed = no }
                        set_warlord_ideologies = yes
                    }
                }
            }
            option = {
                name = "We will not allow this."
                declare_war_on = { target = FROM type = annex_everything }
            }
        }

    # End of Chinese Warlordism?
        news_event = {
            id = manchukuo.19
            title = "End of Chinese Warlordism"
            desc = manchukuo.19.d

            trigger = {
                has_stability > 0.5
                and = { has_idea = idea_warlordism_in_the_countryside has_country_flag = in_warlord_era }
                not = { any_other_country = { is_chinese_warlord = yes exists = yes } }
            }
            fire_only_once = yes

            option = {
                name = "Great."
                remove_ideas = idea_warlordism_in_the_countryside
            }
        }
