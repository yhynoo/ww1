add_namespace = harbin

# different sides claiming things in manchuria
    country_event = {
        id = harbin.1
        title = "[FROM.capital.GetCapitalVictoryPointName] Demands Manchuria"
        desc = harbin.1.d

        is_triggered_only = yes

        option = {
            name = "We must agree."
            if = {
                limit = { is_subject = yes }
                every_country = {
                    limit = { has_subject = ROOT not = { tag = FROM } }
                    every_state = { 
                        limit = { 
                            or = { state = 296 state = 328 state = 619 }
                            is_owned_by = ROOT
                        } 
                        add_claim_by = PREV 
                    }
                    news_event = { id = harbin.2 hours = 6 }
                }
            }

            every_owned_state = {
                limit = { or = { state = 296 state = 328 state = 619 } }
                transfer_state_to = FROM
            }
        }
        option = {
            name = "Outrageous."
            FROM = { create_wargoal = { target = ROOT type = take_claimed_state } }
        }
    }

    news_event = {
        id = harbin.2
        title = "[FROM.FROM.GetAdjective] Troops Enter Manchuria"
        desc = harbin.2.d

        is_triggered_only = yes

        option = {
            name = "This is a hostile act."
            create_wargoal = { target = FROM.FROM type = take_claimed_state }
        }
    }
