            major = yes

            option = {
                name = "Worrying."

        }

            id = HOL_deterrence_event.1

            desc = HOL_deterrence_event.1.d

            trigger = {
                tag = HOL
                and = { has_completed_focus = HOL_deterrence    any_country = { has_wargoal_against = ROOT } }
            }
            fire_only_once = yes

            option = {
                name = "We must do it."
                every_country = {
                    limit = { or = { original_tag = ENG original_tag = FRA } }
                    country_event = { id = guarantees.1 hours = 6 }
                }

                hidden_effect = { random = { chance = 50 news_event = { id = HOL_deterrence_event.2 days = 7 random_days = 7 } } }
            }
