            desc = ROM_aid.1.d

            is_triggered_only = yes

            option = {
                name = "We shall help them."
                add_equipment_to_stockpile = { type = infantry_equipment    amount = -600 }

                # event
                hidden_effect = { news_event = { id = ROM_aid.2 hours = 6 } }
            }
            option = {
                name = "We need this stuff."

                # ai
                ai_chance = { modifier = { is_historical_focus_on = yes } factor = 0 }
            }
        }

                add_war_support = 0.05
