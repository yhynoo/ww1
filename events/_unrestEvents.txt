        id = anti_war.2
        title = "Opposition Demands an Armistice"

        is_triggered_only = yes

        option = {
            name = "It is time to end this war."

            # armistice
            every_country = { limit = { is_armistice_target = yes } country_event = armistice.1 }

        option = {
            name = "We must fight on!"
        }
    }
