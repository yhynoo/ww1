add_namespace = dominion

# AUTONOMY
    # Subject chose the autonomy path
        country_event = {
            id = dominion.1
            title = "[FROM.GetName] Seeks Autonomy?"
            desc = dominion.1.d

            is_triggered_only = yes

            option = {
                name = "Worrying."
                add_political_power = -15
            }
        }

    # Ask for aid
        country_event = {
            id = dominion.2
            title = "[FROM.GetName] Asks for Military Aid"
            desc = dominion.2.d

            is_triggered_only = yes

            option = {
                name = "Sounds good."

                # effect news
                FROM = { news_event = { id = dominion.3 hours = 6 } }
            }
            option = {
                name = "We shall not get involved."
                add_war_support = -0.025
            }
        }

        country_event = {
            id = dominion.3
            title = "Aid from [FROM.GetName] Arrives"
            desc = dominion.3.d

            is_triggered_only = yes

            option = {
                name = "Great."
                add_equipment_to_stockpile = { type = infantry_equipment    amount = 600    producer = ROOT }
                
                # opinion
                add_opinion_modifier = { target = FROM modifier = sent_aid_to_us }
                hidden_effect = { OVERLORD = { add_opinion_modifier = { target = FROM modifier = sent_aid_to_our_subject } } }
            }
        }

    # Subject demands independence
        country_event = {
            id = dominion.4
            title = "[FROM.GetName] Demands Independence!"
            desc = dominion.4.d

            is_triggered_only = yes

            option = {
                name = "We must agree."
                add_war_support = -0.05

                # news
                hidden_effect = { FROM = { news_event = { id = dominion.5 hours = 6 } } }
            }
            option = {
                name = "This means war!"
                declare_war_on = { target = FROM type = annex_everything }

                # news
                hidden_effect = { 
                    FROM = { independence_war_events = yes }
                    every_state = { limit = { is_owned_by = FROM }  add_claim_by = ROOT }
                }
            }
        }

        news_event = {
            id = dominion.5
            title = "[ROOT.GetName] Independent!"
            desc = dominion.5.d

            is_triggered_only = yes
            major = yes

            option = {
                name = "Interesting."
            }
        }

        news_event = {
            id = dominion.6
            title = "[ROOT.GetAdjective] War of Independence"
            desc = dominion.6.d

            is_triggered_only = yes
            major = yes

            option = {
                name = "Alright."
            }
        }

    # Overlord at war - chance?
        country_event = {
            id = dominion.7
            title = "[FROM.GetNameDefCap] at War"
            desc = dominion.7.d

            is_triggered_only = yes
            trigger = { has_civil_war = no }

            option = {
                name = "This is the time."
                every_country = {
                    limit = { has_subject = ROOT }

                    hidden_effect = { set_autonomy = { target = ROOT  autonomous_state = autonomy_free } }
                    ROOT = { declare_war_on = { target = PREV type = independence_wargoal } }
                }
            }
            option = {
                name = "We should sit this one out."
                add_political_power = -50
            }
        }

    # reset tree
        country_event = {
            id = dominion.8
            hidden = yes

            immediate = {
                uncomplete_national_focus = { focus = XXX_autonomy uncomplete_children = yes }
                uncomplete_national_focus = { focus = XXX_allegiance uncomplete_children = yes }
                # mark_focus_tree_layout_dirty = yes
            }
        }

# ALLEGIANCE
    # Subject chose the allegiance path
        country_event = {

            title = "[FROM.GetName] Affirms Their Allegiance"
            desc = dominion.20.d

            is_triggered_only = yes

            option = {
                name = "Great."
                add_political_power = 25
                add_opinion_modifier = { target = FROM modifier = affirmed_their_loyalty }
            }
        }

    # Subject has civil war
        country_event = {
            id = dominion.21
            title = "Intervene in [FROM.GetNameDef]?"
            desc = dominion.21.d

            is_triggered_only = yes

            option = {
                name = "We must act."
                every_country = {
                    limit = { original_tag = FROM not = { is_subject_of = ROOT } }
                    ROOT = { declare_war_on = { target = PREV type = independence_wargoal } }
                }
            }
            option = {
                name = "They will manage on their own."
                add_political_power = -50

                # ai chance
                ai_chance = { modifier = { has_war_support > 0.4 factor = 0.2 } }
            }
        }
