# exchange
    country_event = {
        id = exchange.1
        title = "Organize an Exchange of Territories?"
        desc = exchange.1.d

        trigger = {
            any_allied_country = {
                any_state = {
                    or = { is_core_of = PREV is_claimed_by = PREV } 
                    and = { is_fully_controlled_by = ROOT not = { or = { is_core_of = ROOT is_owned_by = ROOT is_claimed_by = ROOT } } }
                }
            }
        }

        option = {
            name = "Let them take care of it."
            every_allied_country = {
                limit = {
                    any_state = {
                        or = { is_core_of = PREV is_claimed_by = PREV } 
                        and = { is_fully_controlled_by = ROOT not = { or = { is_core_of = ROOT is_claimed_by = ROOT } } }
                    }
                }
                every_state = {
                    limit = {
                        or = { is_core_of = PREV is_claimed_by = PREV } 
                        and = { is_fully_controlled_by = ROOT not = { or = { is_core_of = ROOT is_claimed_by = ROOT } } }
                    }
                    set_state_controller_to = PREV
                }
            }
        }
        option = {
            name = "We will stick to our administration."
            add_political_power = -10

            # ai chance
            ai_chance = { factor = 0 }
        }
    }

        id = warning.1

        trigger = { not = { has_war_with = FROM } }

            name = "This does not concern us."

        }

            name = "We shall issue a warning."

            add_opinion_modifier = { target = FROM modifier = diplomatic_warning }
