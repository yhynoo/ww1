# alliance
    country_event = {
        id = alliance.1
        title = "[FROM.GetName] Offers Alliance"

        is_triggered_only = yes

        option = {
            name = "Great."
            FROM = { 
                custom_effect_tooltip = tt_alliance
                hidden_effect = { add_to_faction = ROOT }
            }

            # ai
            ai_chance = { modifier = { ai_wants_alliance = yes factor = 2 } }
        }
        option = {
            name = "This is too dangerous."
        }
    }

# armistice
    country_event = {
        id = armistice.1
        title = "[FROM.GetName] Offers an Armistice"

        is_triggered_only = yes

        option = {
            name = "Great."
            start_peace_conference = { tag = FROM score_factor = 0.25 }

            # AI + event
            ai_chance = { 
                modifier = { ai_wants_armistice = yes factor = 5 } 
                modifier = { FROM = { has_idea = idea_uprising } factor = 0.25 }
            }
        }
        option = {
            name = "The war shall continue."
            add_political_power = -25
        }
    }

# demands
    country_event = {
        id = demands.1
        title = "[FROM.GetName] Demands [demand_target.GetName]"
        desc = demands.1.d

        is_triggered_only = yes

        option = {
            name = "Let them have it."

            effect_overlord_wargoal = yes
            FROM = { transfer_state = event_target:demand_target }
        }
        option = {
            name = "Outrageous!"
            add_war_support = 0.05

            # defiant
            answer_defiant = yes
        }
    }

    country_event = {
        id = demands.2
        title = "[FROM.GetName] Demands Territories"
        desc = demands.2.d

        is_triggered_only = yes

        option = {
            name = "We must agree."

            effect_overlord_wargoal = yes
            every_owned_state = {
                limit = { is_controlled_by = ROOT or = { is_claimed_by = FROM is_core_of = FROM } }
                transfer_state_to = FROM
            }
        }
        option = {
            name = "We shall not submit to their threats."
            add_war_support = 0.05
            
            # defiant
            answer_defiant = yes

            # ai
            ai_chance = { modifier = { has_idea = idea_war_loser factor = 0 } }
        }
    }

    country_event = {
        id = demands.3
        title = "[FROM.GetName] Demands Alignment!"
        desc = demands.3.d

        is_triggered_only = yes

        option = {
            name = "We must agree."

            effect_overlord_wargoal = yes
            FROM = { set_autonomy = { target = ROOT  autonomous_state = autonomy_dominion    end_wars = no } }
        }
        option = {
            name = "What an insult!"
            add_war_support = 0.05

            # defiant
            answer_defiant = yes
        }
    }
