add_namespace = great_war

    # great war started
        news_event = {
            id = great_war.1
            hidden = yes

            trigger = {
                is_major = yes
                any_enemy_country = {
                    is_major = yes
                    any_other_country = {
                        has_war_together_with = PREV is_major = yes
                    }
                }
            }
            fire_only_once = yes

            immediate = {
                every_country = {
                    limit = { or = { tag = ROOT has_war_with = ROOT has_war_together_with = ROOT } }
                    add_ideas = idea_great_war
                }
            }
        }

    # joins great war
        news_event = {
            id = great_war.2
            hidden = yes

            trigger = {
                not = { has_idea = idea_great_war }
                any_enemy_country = { has_idea = idea_great_war }
            }

            immediate = {
                add_ideas = idea_great_war

                if = {
                    limit = { has_idea = idea_war_loser }
                    remove_ideas = idea_war_loser
                }
            }
        }
