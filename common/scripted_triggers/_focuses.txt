        tooltip = tt_allied_control
        controller = { or = { tag = ROOT       is_ally_with = ROOT } }
    }
}

    custom_trigger_tooltip = {
        tooltip = tt_enemy_control
        controller = { has_war_with = ROOT }
    }
}

has_dominion_focuses = {
    is_subject = yes
}

# unification
has_unification_focuses = {
    check_whether_unified = no
    not = { has_country_flag = assumed_legacy }
}

check_unification_targets = {
    or = {
        and = { 
            ROOT = { or = { original_tag = GRE is_greek_state = yes } }
            and = { not = { tag = ROOT } or = { original_tag = GRE is_greek_state = yes } }
        }

        # default
        and = { not = { tag = ROOT } original_tag = ROOT }
    }
}

check_whether_unified = {
    not = {
        or = {
            and = { is_chinese_state = yes any_other_country = { or = { is_chinese_state = yes } exists = yes } }
            and = { is_greek_state = yes any_other_country = { or = { is_greek_state = yes } exists = yes } }
            and = { is_german_state = yes any_other_country = { or = { is_german_state = yes } exists = yes } }

            # default
            any_other_country = { original_tag = ROOT exists = yes }
        }
    }
}
