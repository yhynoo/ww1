has_dominion_focuses = {
    is_subject = yes
}

has_unification_focuses = {
    or = {
        # Arabia
        is_arab_state = yes

        # China
        is_chinese_warlord = yes

        # Germany
        is_german_state = yes

        # Greece
        is_greek_state = yes
    }
    not = { has_country_flag = completed_unification }

    if = {
        limit = { can_choose_ambition = yes }
        has_country_flag = chose_unifier_identity
    }
}

# civil war
    civil_war_industry_lock = {
        if = { 
            limit = { has_country_flag = during_civil_war }
            if = {
                limit = { not = { has_idea = idea_eroded_state } }
                custom_trigger_tooltip = {
                    tooltip = tt_civil_war_lock
                    and = { has_country_flag = during_civil_war has_completed_focus = XXX_wheat_and_lead }
                }
            }
            else = { 
                custom_trigger_tooltip = { tooltip = tt_eroded_state    always = no }
            }
        }
    }

    civil_war_army_lock = {
        if = { 
            limit = { has_country_flag = during_civil_war }
            if = {
                limit = { not = { has_idea = idea_eroded_state } }
                custom_trigger_tooltip = {
                    tooltip = tt_civil_war_lock
                    and = { has_country_flag = during_civil_war has_completed_focus = XXX_local_commands }
                }
            }
            else = { 
                custom_trigger_tooltip = { tooltip = tt_eroded_state    always = no }
            }
        }
    }

    civil_war_government_lock = {
        if = { 
            limit = { has_country_flag = during_civil_war }
            if = {
                limit = { not = { has_idea = idea_eroded_state } }
                custom_trigger_tooltip = {
                    tooltip = tt_civil_war_lock
                    and = { has_country_flag = during_civil_war has_completed_focus = XXX_united_again }
                }
            }
            else = { 
                custom_trigger_tooltip = { tooltip = tt_eroded_state    always = no }
            }
        }
    }

    civil_war_erosion_lock = {
        if = {
            limit = { has_idea = idea_eroded_state }
            custom_trigger_tooltip = { tooltip = tt_eroded_state    always = no }
        }
    }

# uprising
    uprising_government_lock = {
        if = {
            limit = { has_country_flag = during_uprising }
            custom_trigger_tooltip = {
                tooltip = tt_interim_government_lock
                not = { has_country_flag = during_uprising }
            }
        }
    }
