# civil war
    civil_war_lock = {
        if = { 
            limit = { has_idea = idea_eroded_state }
            custom_trigger_tooltip = { tooltip = tt_eroded_state    always = no }
        }
    }

# colonies: do we own any distant states that are not our cores?
    colonies_lock = {
        custom_trigger_tooltip = {
            tooltip = tt_colonies_lock
            any_owned_state = { is_colonial_state = yes is_controlled_by = ROOT }
        }
    }

    is_colonial_state = {
        not = { is_core_of = ROOT }
        or = {
            is_on_continent = africa
            and = { 
                ROOT.capital_scope = { is_on_continent = europe } 
                is_on_continent = asia 
            }
        }
    }
