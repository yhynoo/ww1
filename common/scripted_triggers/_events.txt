ai_wants_alliance = {
    or = {
        has_opinion = { target = FROM value > 25 }
        and = { 
            has_opinion = { target = FROM value > 10 } 
            threat > 0.4 
        }
    }
}

ai_wants_armistice = {
    or = {
        has_idea = idea_war_exhaustion
        surrender_progress > 0.05
        has_war_support < 0.25
    }
}

can_have_unrest_event = {
    not = { has_idea = idea_unrest_timeout }
    has_stability < 0.4
}

can_have_ideology_event = {
    not = { has_country_flag = during_uprising }
    not = { has_idea = idea_ideology_timeout }

    has_civil_war = no
    or = {
        and = { check_variable = { highest_party_popularity@exclude_ruling_party > 0.25 } stability < 0.4 }
        check_variable = { highest_party_popularity@exclude_ruling_party > 0.4 }
    }
}

can_have_government_change_event = {
    check_variable = { highest_party_popularity@exclude_ruling_party > 0.5 }
    not = { has_country_flag = during_uprising }
    has_civil_war = no
}

can_have_wheat_affair = {
    longest_war_length > 24
    or = { has_idea = partial_economic_mobilisation has_idea = war_economy }
    not = { has_idea = idea_war_exhaustion }
}

can_have_opposition_event = {
    is_subject = no
    or = {
        and = { has_war = yes has_war_support < 0.4 longest_war_length > 11 }
        surrender_progress > 0.15
    }
    any_enemy_country = { is_armistice_target = yes }

    check_variable = { highest_party_popularity@exclude_ruling_party > 0.25 }
    not = { has_idea = idea_armistice_timeout }
}

can_fall_apart = {
    or = { 
        has_civil_war = yes 
        or = {
            and = { surrender_progress > 0.1 has_stability < 0.6 }
            and = { surrender_progress > 0.25 has_stability < 0.8 }
        }
        and = { has_idea = idea_war_loser has_stability < 0.45 }
        has_stability < 0.2 
    }
}

can_trigger_uprising = {
    exists = no 
    has_capitulated = no 
    not = { has_country_flag = country_defeated }
    any_core_state = { is_controlled_by = ROOT }
}

is_armistice_target = {
    and = { has_war_with = ROOT not = { original_tag = ROOT } }     # not our civil war enemies.
    is_diplomatically_able = yes

    # block chinese warlords
    is_chinese_warlord = no
}

is_diplomatically_able = {
    or = { is_faction_leader = yes is_in_faction = no }
    is_subject = no
}

is_guaranteeable = {
    is_diplomatically_able = yes
    and = {
         not = { is_guaranteed_by = FROM }
         not = { has_war_with = FROM }
    }
}

has_custom_events = {
    or = { 
        or = { }    # list of tags.
    }       
}
