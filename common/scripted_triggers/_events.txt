
has_custom_events = {
    or = { }       # list of tags.
    always = no
}

can_have_unrest_event = {
    not = { has_idea = idea_unrest_timeout }
    and = { check_variable = { highest_party_popularity@exclude_ruling_party > 0.05 } stability < 0.4 }
}

can_have_ideology_event = {
    not = { has_country_flag = during_uprising }
    not = { has_idea = idea_ideology_timeout }

    has_civil_war = no
    or = {
        and = { check_variable = { highest_party_popularity@exclude_ruling_party > 0.2 } stability < 0.5 }
        and = { check_variable = { highest_party_popularity@exclude_ruling_party > 0.25 } stability < 0.65 }
    }

    date > 1912.2.1
}

can_have_government_change_event = {
    not = { has_country_flag = during_uprising }
    has_civil_war = no

    or = {
        and = {
            or = { 
                check_variable = { highest_party_ideology@exclude_ruling_party = token:communism }
                check_variable = { highest_party_ideology@exclude_ruling_party = token:monarchism }
            }
            or = {
                and = { 
                    not = { has_idea = idea_coup_timeout }
                    or = {
                        and = { check_variable = { highest_party_popularity@exclude_ruling_party > 0.45 } stability < 0.5 }
                        and = { check_variable = { highest_party_popularity@exclude_ruling_party > 0.55 } stability < 0.65 }
                    }
                }
                and = {
                    has_idea = idea_coup_timeout
                    check_variable = { highest_party_popularity@exclude_ruling_party > 0.25 } stability < 0.1
                }
            }
        }
        and = {
            check_variable = { highest_party_ideology@exclude_ruling_party = token:democratic }
            check_variable = { highest_party_popularity@exclude_ruling_party > 0.5 }
        }
    }
}
