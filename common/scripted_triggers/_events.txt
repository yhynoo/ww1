ai_wants_alliance = {
    or = {
        has_opinion = { target = FROM value > 25 }
        and = { 
            has_opinion = { target = FROM value > 10 } 
            threat > 0.4 
        }
    }
}

ai_wants_armistice = {
    or = {
        surrender_progress > 0.15
        has_war_support < 0.4
        has_stability < 0.25

        FROM = { or = { is_major = yes is_faction_leader = yes } }
    }
}

can_have_armistice_event = {
    is_subject = no
    has_civil_war = no
    any_enemy_country = { is_armistice_target = yes }

    or = {
        and = { longest_war_length > 24     has_war_support < 0.25 }
        and = { longest_war_length > 6      or = { has_war_support < 0.15 surrender_progress > 0.15 } }
    }

    not = { has_idea = idea_armistice_timeout }
}

is_armistice_target = {
    and = { has_war_with = ROOT not = { original_tag = ROOT } }     # not our civil war enemies.
    is_diplomatically_able = yes

    # block chinese warlords
    is_chinese_state = no
}

# unused
    can_have_ideology_event = {
        not = { has_country_flag = during_uprising }
        not = { has_idea = idea_ideology_timeout }

        has_civil_war = no
        or = {
            and = { check_variable = { highest_party_popularity@exclude_ruling_party > 0.25 } stability < 0.4 }
            check_variable = { highest_party_popularity@exclude_ruling_party > 0.4 }
        }
    }

    can_fall_apart = {
        or = { 
            has_civil_war = yes 
            or = {
                and = { surrender_progress > 0.1 has_stability < 0.6 }
                and = { surrender_progress > 0.25 has_stability < 0.8 }
            }
            and = { has_idea = idea_war_loser has_stability < 0.8 }
            has_stability < 0.2 
        }
    }

    can_trigger_uprising = {
        exists = no 
        has_capitulated = no
        is_dynamic_country = no
        
        not = { has_country_flag = country_defeated has_idea = idea_war_loser }
        capital_scope = { is_fully_controlled_by = ROOT }

        not = {
            and = {                                 # germany won't fall apart unless in civil war.
                is_german_state = yes
                GER = { has_civil_war = no }
            }
        }
    }

    is_diplomatically_able = {
        or = { is_faction_leader = yes is_in_faction = no }
        is_subject = no
    }

    is_guaranteeable = {
        is_diplomatically_able = yes
        and = {
            not = { is_guaranteed_by = FROM }
            not = { has_war_with = FROM }
        }
    }

    has_custom_events = {
        or = { 
            or = { }    # list of tags.
        }       
    }
