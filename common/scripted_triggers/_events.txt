ai_wants_alliance = {
    or = {
        has_opinion = { target = FROM value > 25 }
        and = { 
            has_opinion = { target = FROM value > 10 } 
            threat > 0.4 
        }
    }
}

ai_wants_armistice = {
    or = {
        # main condition
            and = {
                or = { surrender_progress > 0.1     has_war_support < 0.15                      }
                or = { has_stability < 0.25         has_idea = idea_anti_war_demonstrations     }
            }

        # southeastern_europe_war
            and = {
                is_historical_focus_on = yes
                has_global_flag = southeastern_europe_war
                
                or = { original_tag = ARM original_tag = GRE original_tag = ROM original_tag = SER original_tag = TUR }
            }
    }
}

can_have_an_uprising = {
    has_special_collapse_sequence = no
    or = { 
        has_civil_war = yes 
        or = {
            and = { surrender_progress > 0.1 has_stability < 0.25 }
            and = { surrender_progress > 0.25 has_stability < 0.5 }
        }
    }
}

can_have_a_civil_war = {
    has_special_collapse_sequence = no
    has_civil_war = no
    check_variable = { highest_party_popularity@exclude_ruling_party > 0.2 }

    or = {
        has_stability < 0.1
        and = {
            has_stability < 0.4
            or = {
                and = { surrender_progress > 0.1        has_war_support < 0.25 }
                and = { surrender_progress > 0.2        has_war_support < 0.4 }
            }
        }
    }
}

can_start_uprising = {
    exists = no     has_capitulated = no        is_dynamic_country = no 
    capital_scope = { is_controlled_by = ROOT }
}

has_special_collapse_sequence = {
    or = {
        has_country_flag = austria_house_collapse
        has_country_flag = wheat_deal_collapse
    }
}

is_armistice_target = {
    and = { has_war_with = ROOT not = { original_tag = ROOT } }     # not our civil war enemies.
    is_diplomatically_able = yes

    # block chinese warlords
    is_chinese_state = no
}

is_diplomatically_able = {
    or = { is_faction_leader = yes is_in_faction = no }
    is_subject = no
}

is_guaranteeable = {
    is_diplomatically_able = yes
    and = {
         not = { is_guaranteed_by = FROM }
         not = { has_war_with = FROM }
    }
}
