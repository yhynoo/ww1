    is_defensive = {
        or = {
            and = { original_tag = FRA not = { has_government = monarchism } }
            has_country_flag = HOL_democratic
        }
    }

    is_ideologically_committed = {
        or = {
            is_subject = yes
            and = {
                check_variable = { highest_party_ideology@exclude_ruling_party = token:communism }
                or = { is_going_democrat = yes is_going_monarchist = yes }
            }
            and = {
                check_variable = { highest_party_ideology@exclude_ruling_party = token:democratic }
                or = { is_going_communist = yes is_going_monarchist = yes }
            }
            and = {
                check_variable = { highest_party_ideology@exclude_ruling_party = token:monarchism }
                or = { is_going_communist = yes is_going_democrat = yes }
            }
        }
    }
