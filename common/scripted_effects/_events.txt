answer_defiant = {
    FROM = {
        country_event = { id = defiant.1 hours = 6 }
        custom_effect_tooltip = tt_defiant
    }
}

answer_warning = {
    hidden_effect = { country_event = { id = warning.1 days = 2 } }
}

independence_war_events = {
    news_event = { id = dominion.6 hours = 6 }
}

change_government = {
    if = {
        limit = { communism > 0.5 }
        set_politics = { ruling_party = communism elections_allowed = no }
    }
    if = {
        limit = { democratic > 0.5 }
        set_politics = { ruling_party = democratic elections_allowed = yes }
    }
    if = {
        limit = { monarchism > 0.5 }
        set_politics = { ruling_party = monarchism elections_allowed = no }
    }
}

change_government_opposed = {
    if = {
        limit = { communism > 0.5 }
        start_civil_war = { ideology = communism size = 0.5 }
    }
    if = {
        limit = { democratic > 0.5 }
        start_civil_war = { ideology = democratic size = 0.5 }
    }
    if = {
        limit = { monarchism > 0.5 }
        start_civil_war = { ideology = monarchism size = 0.5 }
    }
}

opposition_idea = {
    if = {
        limit = { check_variable = { highest_party_ideology@exclude_ruling_party = token:communism } }
        add_timed_idea = { idea = idea_communists_organize months = 24 }
    }
    if = {
        limit = { check_variable = { highest_party_ideology@exclude_ruling_party = token:democratic } }
        add_timed_idea = { idea = idea_democrats_organize months = 24 }
    }
    if = {
        limit = { check_variable = { highest_party_ideology@exclude_ruling_party = token:monarchism } }
        add_timed_idea = { idea = idea_monarchists_organize months = 24 }
    }
}

leave_alliances = {
    if = {
        limit = { is_faction_leader = yes }
        dismantle_faction = yes
    }
    else = {
        if = {
            limit = { is_in_faction = yes }
            leave_faction = yes
        }
    }
    every_country = {
        limit = { is_guaranteed_by = ROOT }
        ROOT = { diplomatic_relation = { country = PREV relation = guarantee active = no } }
    }
    every_country = {
        limit = { has_non_aggression_pact_with = ROOT }
        ROOT = { diplomatic_relation = { country = PREV relation = non_aggression_pact active = no } }
    }
}

leave_alliances_answer = {
    if = {
        limit = { has_guaranteed = FROM }
        diplomatic_relation = { country = FROM relation = guarantee active = no }
    }
    if = {
        limit = { has_non_aggression_pact_with = FROM }
        diplomatic_relation = { country = FROM relation = non_aggression_pact active = no }
    }
}
