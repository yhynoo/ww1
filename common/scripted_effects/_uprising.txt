trigger_uprising = {
    setup_uprising_country = yes 
    # uprising_reaction_event = yes
}

setup_uprising_country = {
    # assign states
    every_core_state = {
        limit = { and = { owner = { original_tag = ROOT } controller = { original_tag = ROOT } } }
        transfer_state_to = PREV

        # manage states
        if = {
            limit = { is_capital = no }
            random = { chance = 40 set_state_controller_to = ROOT }

            if = {
                limit = { not = { any_neighbor_state = { is_capital = yes } } }
                random = { chance = 80 set_state_controller_to = ROOT }
            }
        }
    }

    # setup specific states
    every_controlled_state = { setup_uprising_state = yes }

    # flags and focuses
    if = {
        limit = { not = { has_country_flag = started_as_uprising } }
        set_country_flag = started_as_uprising
        set_country_flag = during_uprising
        
        add_timed_idea = { idea = idea_uprising months = 6 }
    }

    declare_war_on = { target = PREV type = independence_wargoal }
}

uprising_reaction_event = {
    save_event_target_as = uprising_country
    every_country = {
        limit = {
            not = { tag = PREV }
            or = {
                tag = PREV.PREV
                any_state = {
                    is_owned_by = event_target:uprising_country
                    or = { 
                        is_core_of = PREV 
                        is_claimed_by = PREV 
                    }
                }
            }
        }
        country_event = uprising.1
    }
}

setup_uprising_state = {
    owner = { setup_uprising_troops = yes }
    create_unit = { division = "name = \"Uprising Troops\" division_template = \"Uprising Troops\"" count = 2 owner = PREV }

    random = {
        chance = 60
        create_unit = { division = "name = \"Uprising Troops\" division_template = \"Uprising Troops\"" count = 1 owner = PREV }
    }
}

setup_guards_units = {
    if = {
        limit = { not = { has_template = "Guards Division" } }
        division_template = {
            name = "Guards Division"
            is_locked = yes
            regiments = {
                cavalry = { x = 0 y = 0 }      cavalry = { x = 1 y = 0 }
                cavalry = { x = 0 y = 1 }      cavalry = { x = 1 y = 1 }
            }
        }
    }
}
