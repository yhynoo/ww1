trigger_uprising = {
    every_core_state = {
        limit = { controller = { original_tag = ROOT } }
        set_variable = { original_controller = controller }     transfer_state_to = PREV

        # set up
        if = { limit = { is_capital = yes } setup_uprising_volunteers = yes setup_uprising_volunteers = yes }       # twice to make sure there is more
        if = { 
            limit = { is_capital = no }  
            random_list = {
                1 = { random = { chance = 50 setup_uprising_volunteers = yes } }
                1 = { random = { chance = 50 every_country = { limit = { original_tag = ROOT } transfer_units_fraction = { target = PREV.PREV      size = 0.025     stockpile_ratio = 0.05     keep_unit_leaders_trigger = { always = no } } } } }
            }
        }
    }

    save_event_target_as = uprising_country 
    every_country = {
        limit = { 
            not = { tag = PREV } 
            and = { any_state = { is_owned_by = PREV.PREV or = { is_core_of = PREV is_claimed_by = PREV } } }
        }
        country_event = uprising.1
    }

    # control
    every_owned_state = { var:original_controller = { set_state_controller = PREV } }

    # flags and focuses
    add_ideas = idea_uprising
}

setup_uprising_volunteers = {
    owner = { division_template = { name = "Uprising Troops" regiments = { infantry = { x = 0 y = 0 } infantry = { x = 1 y = 0 } } } }
    random_list = {
        2 = { create_unit = { division = "name = \"Uprising Troops\" division_template = \"Uprising Troops\"" count = 2 owner = PREV } }
        1 = { create_unit = { division = "name = \"Uprising Troops\" division_template = \"Uprising Troops\"" count = 1 owner = PREV } }
    }
}
