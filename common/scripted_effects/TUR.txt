manage_turkish_debt = {
    custom_effect_tooltip = tt_manage_turkish_debt
    hidden_effect = {
        if = {
            limit = { has_idea = idea_amassed_foreign_debt }
            swap_ideas = {
                remove_idea = idea_amassed_foreign_debt
                add_idea = idea_limited_foreign_debt
            }
        }
        else = {
            remove_ideas = idea_limited_foreign_debt
        }
    }
}
