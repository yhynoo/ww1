manage_turkish_debt = {
    custom_effect_tooltip = tt_decrease_debt
    hidden_effect = {
        if = {
            limit = {
                has_idea = idea_amassed_foreign_debt
            }
            swap_ideas = {
                remove_idea = idea_amassed_foreign_debt 
                add_idea = idea_limited_foreign_debt
            }
        }
        else = {
            remove_ideas = idea_limited_foreign_debt
        }
    }
}

manage_turkish_officers = {
    custom_effect_tooltip = tt_decrease_officers
    hidden_effect = {
        if = {
            limit = {
                has_idea = idea_hostile_officer_corps
            }
            swap_ideas = {
                remove_idea = idea_hostile_officer_corps 
                add_idea = idea_disobedient_officer_corps
            }
        }
        else = {
            remove_ideas = idea_disobedient_officer_corps
        }
    }
}
