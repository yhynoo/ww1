effect_overlord_warning_country = {
    if = { 
        limit = { any_other_country = { has_subject = PREV } }
        custom_effect_tooltip = tt_overlord_wargoal
    }
}

effect_overlord_warning_state = {
    if = { 
        limit = { any_other_country = { has_subject = PREV.owner } }
        custom_effect_tooltip = tt_overlord_wargoal
    }
}

effect_overlord_wargoal = {
    if = {
        limit = { any_other_country = { has_subject = PREV } }
        hidden_effect = {
            every_country = {
                limit = { has_subject = ROOT not = { tag = FROM } }
                
                # create_wargoal = { target = ROOT type = puppet_wargoal_focus }
                create_wargoal = { target = FROM type = liberate_wargoal }
            }
        }
    }
}

effect_warlike_option = {
    if = {
        limit = { has_war_support > 0.4 }
        add_war_support = 0.05
    }
    else = {
        add_political_power = -15
    }
}

effect_quiet_option = {
    if = {
        limit = { has_stability < 0.4 }
        add_stability = 0.025
    }
    else = {
        add_war_support = -0.025
    }
}

manage_client_state_tags = {
    # china
        if = {
            limit = { 
                or = { original_tag = MAN is_chinese_state = yes } has_war_with = MAN 
                not = { original_tag = FER }
            }
            set_cosmetic_tag = MAN_enemy_warlord
        }
        if = {
            limit = { not = { has_war_with = MAN } has_cosmetic_tag = MAN_enemy_warlord }
            drop_cosmetic_tag = yes
        }
        if = {
            limit = { has_cosmetic_tag = MAN_empire not = { has_government = monarchism } }
            drop_cosmetic_tag = yes
        }

    # cleanup of india tag
        if = {
            limit = { original_tag = RAJ is_subject = no }
            drop_cosmetic_tag = yes
        }
}
