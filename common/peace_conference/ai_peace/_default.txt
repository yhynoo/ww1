peace_ai_desires = {

    # TAKING STATES
    take_cores = {
        peace_action_type = { take_states }
        enable = { FROM.FROM.FROM = { is_core_of = ROOT } }        # first, just grab cores and neighboring states.
        ai_desire = 200
    }

    take_around = {
        peace_action_type = { take_states }
        enable = {
            FROM.FROM.FROM = {
                or = { is_claimed_by = ROOT is_controlled_by = ROOT }
            }
        }
        ai_desire = 30
    }

    take_farther = {
        peace_action_type = { take_states }
        enable = {
            FROM.FROM.FROM = {
                if = {
                    limit = { ROOT = { is_major = no } }        # if we're not majors, we're not interested in africa
                    is_on_continent = africa
                }
                not = { 
                    or = {
                        is_controlled_by = ROOT 
                        is_claimed_by = ROOT
                        is_core_of = ROOT
                    }
                }
            }
        }
        ai_desire = -200
    }

    dont_take_nonsens = {
        peace_action_type = { take_states }
        enable = {
            FROM.FROM.FROM = {
                or = {
                    not = {
                        any_neighboring_state = {
                            is_owned_by = ROOT
                            pc_is_state_claimed_and_taken_by = ROOT
                        }
                    }
                    is_coastal = no
                }
            }
        }
        ai_desire = -600
    }
    
    # DOMINIONS
    keep_them_tight = {
        peace_action_type = { puppet }
        enable = {
            FROM.FROM.FROM = {
                any_neighboring_state = { pc_is_puppeted_by = ROOT }
            }
        }
        ai_desire = 30
    }

    dont_take_states_from_dominions = {
        peace_action_type = { take_state }
        enable = { FROM = { pc_is_puppeted_by = ROOT } }
        ai_desire = -100
    }

    # OTHER
    dont_do_other_things = {
        peace_action_type = { puppet liberate force_government }
        enable = { always = yes }
        ai_desire = -600
    }
}
