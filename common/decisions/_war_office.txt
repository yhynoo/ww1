_war_office = {
    inspire_definitive_offensive = {
        visible = { 
            has_war = yes
            has_war_support > 0.25
            not = { surrender_progress > 0.05 }
        }

        icon = generic_ignite_civil_war
        days_remove = 180
        cost = 100

        modifier = {
            army_attack_factor = 0.15
            army_morale_factor = 0.2
        }

        remove_effect = {
            add_war_support = -0.25
            add_popularity = { ideology = ROOT popularity = -0.05 }
        }

        cancel_trigger = { has_war = no }
    }
# mobilization
    extend_defenses = {
        icon = ger_military_buildup
        days_remove = 90
        cost = 75

        available = {
            custom_trigger_tooltip = {
                tooltip = tt_mobilization_decision
                has_country_flag = mobilization_decisions_allowed
            }
        }

        modifier = {
            production_speed_bunker_factor = 0.25
            max_dig_in_factor = 0.2
        }
    }
    organize_timetables = {
        icon = generic_prepare_civil_war
        days_remove = 90
        cost = 50

        available = {
            has_tech = basic_train
            custom_trigger_tooltip = {
                tooltip = tt_mobilization_decision
                has_country_flag = mobilization_decisions_allowed
            }
        }

        ai_will_do = { modifier = { has_war = yes factor = 6 } }

        modifier = {
            supply_factor = 0.15
            army_org_factor = 0.1
        }
    }

# territories
    territorial_adjustments = {             # if we're fighting against the same enemy, we can reorganize our territories to do it more effectively.
        visible = { has_war = yes }

        icon = border_war
        days_re_enable = 90
        cost = 50

        targets_dynamic = yes
        target_trigger = {
            FROM = {
                not = { has_war_with = ROOT }
                not = { any_enemy_country = { original_tag = FROM } }       # not civil war sides.
                any_enemy_country = { has_war_with = ROOT }
                is_chinese_warlord = no

                or = {
                    any_state = { 
                        and = { is_controlled_by = FROM not = { is_claimed_by = FROM } not = { is_owned_by = FROM } }
                        is_core_of = ROOT
                    }
                    any_state = {
                        and = { is_controlled_by = ROOT not = { is_claimed_by = ROOT } not = { is_owned_by = ROOT } }
                        is_core_of = FROM
                    }
                }
            }
        }

        ai_will_do = { factor = 6 }

        days_remove = 7
        remove_effect = { FROM = { country_event = diplomacy.6 } }
    }

# armistice
    withdraw_from_war = {                   # if the war has lasted long enough (at least 6 months)
        visible = { 
            has_war = yes 
            is_subject = no
            war_length_with = { tag = FROM months > 5 }
        }

        icon = generic_form_nation
        days_re_enable = 90
        cost = 100

        targets_dynamic = yes
        target_trigger = { 
            FROM = {
                has_war_with = ROOT
                not = { original_tag = ROOT }       # not our civil war enemies.
                or = { 
                    is_faction_leader = yes 
                    is_in_faction = no
                }
                is_subject = no

                # block chinese warlords
                is_chinese_warlord = no
            }
        }

        ai_will_do = {
            modifier = { has_war_support < 0.5 add = 4 }    # entry condition: if we have more war support, we won't even think about it.
            modifier = { or = { has_idea = idea_war_exhaustion has_stability < 0.2 } factor = 2 }
            modifier = { surrender_progress > 0.15 factor = 1.5 }

            # if we're done, we're done.
            modifier = { has_capitulated = yes add = 12 }
        }

        days_remove = 14
        remove_effect = { 
            FROM = { country_event = { id = diplomacy.8 hours = 6 random_hours = 6 } } 
        }
    }

    demand_surrender = {
        visible = { 
            has_war = yes 
            is_subject = no

            or = {
               is_faction_leader = yes
               is_in_faction = no
            }

            FROM = { surrender_progress > 0.05 }
            war_length_with = { tag = FROM months > 2 }
        }

        icon = generic_public_support
        days_re_enable = 90
        cost = 75

        available = { 
            FROM = {
                or = { 
                    surrender_progress > 0.15 
                    has_war_support < 0.25 
                }
            }
        }

        ai_will_do = {
            modifier = { always = yes add = 4 }     # if we can, there is a chance we will do it.
            modifier = { or = { has_idea = idea_war_exhaustion has_war_support < 0.6 has_stability < 0.4 } factor = 3 }   # if we sort of want out of it, we will ask more certainly.
            modifier = { FROM = { surrender_progress > 0.6 } factor = 0.2 } # if we're close to winning, we might as well just finish the job.
        }

        targets_dynamic = yes
        target_trigger = { 
            FROM = {
                has_war_with = ROOT
                not = { original_tag = ROOT }       # not our civil war enemies.

                or = { 
                    is_faction_leader = yes 
                    is_in_faction = no
                }
                is_subject = no

                # block chinese warlords
                is_chinese_warlord = no
            }
        }

        days_remove = 14
        remove_effect = { 
            FROM = { country_event = { id = diplomacy.17 hours = 6 random_hours = 6 } } 
        }
    }
}
