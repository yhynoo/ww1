_arab_unification = {
    assume_arab_legacy = {
        icon = generic_nationalism

        available = { 
            has_completed_focus = XXX_destiny
        }
        ai_will_do = { add = 6 }

        complete_effect = {
            set_country_flag = completed_unification
            mark_focus_tree_layout_dirty = yes
            
            # event to choose capital
            remove_ideas = idea_united_in_destiny
        }
    }
    demand_arab_submit = {
        icon = generic_prepare_civil_war
        fire_only_once = yes
        
        targets_dynamic = yes
        target_trigger = {
            FROM = {
                not = { has_war_with = ROOT }
                or = { 
                    original_tag = ASY
                    is_arab_state = yes 
                }
            }
        }

        cost = 75
        ai_will_do = {
            modifier = { 
                or = {
                    FROM = { is_subject = no }
                    any_country = { limit = { has_subject = FROM } has_war = yes }
                }
                factor = 2
            }
            modifier = {
                has_completed_focus = XXX_ethnocentrism
                add = 2
            }
            modifier = {
                strength_ratio = { tag = FROM ratio > 1.5 }
                add = 4
            }
        }

        complete_effect = {
            every_state = {
                limit = { 
                    is_owned_by = FROM 
                    not = { or = { is_claimed_by = ROOT is_core_of = ROOT } }
                }
                add_claim_by = ROOT
            }

            FROM = {
                country_event = { id = unification.2 hours = 6 }
                effect_overlord_warning = yes
            }
        }
    }
}
