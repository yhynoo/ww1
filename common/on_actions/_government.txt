on_actions = {
    on_ruling_party_change = {
        effect = {
            if = {
                limit = { has_government = communism }
                load_focus_tree = communist_focus

                if = {
                    not = { or = { has_cosmetic_tag = RUS_soviet has_cosmetic_tag = UKR_soviet } }
                    set_cosmetic_tag = generic_communism
                }
            }
        }
    }

    on_subject_free = {
        effect = {
            mark_focus_tree_layout_dirty = yes
        }
    }
}
