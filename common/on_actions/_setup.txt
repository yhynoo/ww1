on_actions = {
    on_startup = {
        effect = {

        # resource rights
            COL = {
                give_resource_rights = { receiver = USA state = 312 }
                give_resource_rights = { receiver = USA state = 317 }
            }

            MEX = {
                give_resource_rights = { receiver = USA state = 476 }
                give_resource_rights = { receiver = USA state = 485 }
            }

            QAJ = {
                give_resource_rights = { receiver = ENG state = 412 }
                give_resource_rights = { receiver = ENG state = 413 }
            }

        # events
            AUS = {
                news_event = { id = emperor.1 months = 46 days = 27 }
            }
            
            ENG = {
                news_event = { id = titanic.1 months = 3 days = 14 }
            }

            FRA = {
                news_event = { id = garros.1 months = 20 days = 3 }
            }

            GER = {
                news_event = { id = equestrian.1 days = 26 }
                news_event = { id = johannisthal.1 months = 21 days = 17 }
            }

            GRE = {
                news_event = { id = konstantinos.1 months = 14 days = 7 }
            }

            HOL = {
                news_event = { id = soesterberg.1 months = 7 days = 14 }
            }

            ITA = {
                news_event = { id = manifesto_intonarumori.1 months = 17 days = 5 }
            }

            JAP = {
                news_event = { id = meiji.1 months = 6 days = 30 }
                country_event = { id = diet.1 months = 12 days = 7 }
            }

            MAN = {
                country_event = { id = tibet.1 months = 3 days = 4 }
            }

            MEX = {
                news_event = { id = mexico_uprising.1 months = 1 days = 2 }
            }

            RUS = {
                news_event = { id = muromets.1 months = 22 days = 6 }
            }

            TUR = {
                news_event = { id = aircraft_ottomans.1 months = 2 days = 20 }
            }

            USA = {
                # Act I:
                news_event = { id = unions_ludlow.1 months = 27 days = 20 }

                # Act III:
                news_event = { id = harvest.1 months = 68 }
                news_event = { id = wheat_shortage.1 months = 72 days = 14 }
            }
        }
    }

    on_daily = {
        effect = {
            # clean uprising tags
                if = {
                    limit = { 
                        has_idea = idea_uprising
                        or = { has_completed_focus = XXX_interim_government has_war = no }
                    }
                    country_event = other.3
                }       

            # warlord colors
                if = {
                    limit = { or = { is_chinese_warlord = yes original_tag = MAN } has_war_with = MAN }
                    set_cosmetic_tag = MAN_enemy_warlord
                }
                if = {
                    limit = { has_cosmetic_tag = MAN_enemy_warlord not = { has_war_with = MAN } }
                    drop_cosmetic_tag = yes
                }

            # autonomy loss
                if = {
                    limit = { is_subject = yes has_elections = yes }
                    set_politics = { ruling_party = ROOT elections_allowed = no }
                }

                if = {
                    limit = { has_cosmetic_tag = AUH    HUN = { exists = yes } }
                    drop_cosmetic_tag = yes
                }
                if = {
                    limit = { or = { original_tag = HUN original_tag = CRO } is_subject_of = AUS not = { has_cosmetic_tag = austrian_client_state } }
                    set_cosmetic_tag = austrian_client_state
                }
            
            #
        }
    }

    on_weekly = {
        effect = {
        # uprisings
            if = {
                limit = { 
                    can_have_an_uprising = yes 
                    not = { has_cosmetic_tag = AUH }        # austria-hungary has special collapse
                }
                every_possible_country = { 
                    limit = { can_start_uprising = yes } 
                    random = { 
                        chance = 5 
                        trigger_uprising_easy = yes     save_event_target_as = uprising_country         news_event = uprising.1
                    } 
                }
            }

        # civil wars
            if = {
                limit = { 
                    can_have_a_civil_war = yes 
                    not = { has_cosmetic_tag = AUH }            # austria-hungary has special collapse
                    not = { has_cosmetic_tag = MEX_madero }     # mexico has its own civil war
                }
                random = { chance = 10 trigger_civil_war = yes }
            }

        # airships
            if = {
                limit = { any_owned_state = { and = { days_since_last_strategic_bombing < 7 not = { has_global_flag = airships_used } } } }
                news_event = airships.1
            }

        # gas
            if = {
                limit = { has_war = yes has_tech = gas_shells not = { has_global_flag = gas_used } }
                news_event = gas.1
            }

            # country specifics
                if = {
                    limit = { 
                        tag = RUS 
                        and = { communism > 0.25     has_stability < 0.4 } 
                    }
                    country_event = astrakhan.1
                }
        }
    }

    on_weekly_AUS = {
        effect = {
            if = {
                limit = { has_completed_focus = generic_oil has_stability < 0.8 }
                country_event = galician_oil_tensions.1
            }
            if = {
                limit = {
                    has_cosmetic_tag = AUH
                    or = {
                        and = { has_idea = idea_wheat_shortages or = { has_idea = idea_ethnic_tensions has_idea = idea_ethnic_tensions_1 } }
                        POL = { exists = yes is_subject = no has_war = no }

                        can_have_a_civil_war = yes
                        can_have_an_uprising = yes
                    }
                }
                random = { chance = 20      country_event = house_manifest.1 }
            }
        }
    }

    on_weekly_ENG = {
        effect = {
            if = {
                limit = { has_completed_focus = XXX_extend }
                if = {
                    limit = {
                        FRA = { has_war_with = GER  or = { surrender_progress > 0.15 29 = { is_controlled_by = GER } } }
                        can_declare_war_on = GER
                    }
                    country_event = admiralty_channel.1   
                }
            }

            if = {
                limit = {
                    longest_war_length > 12
                    has_war_support < 0.65      not = { has_idea = volunteer_only }
                }
                country_event = conscription_crisis.1
            }
        }
    }

    on_weekly_GER = {
        effect = {
            if = {
                limit = { RUS = { has_country_flag = wheat_deal_collapse } POL = { exists = yes } UKR = { exists = yes } }
                country_event = kadets_ohl.1
            }
        }
    }

    on_weekly_ITA = {
        effect = {
            if = {
                limit = { has_idea = idea_trauma_of_tripoli     FRA = { has_war_with = AUS } }
                news_event = manifesto_avanti.1
            }
        }
    }

    on_weekly_POL = {
        effect = {
            if = {
                limit = { is_guaranteed_by = USA    has_war = no }
                country_event = kadets_washington.3
            }
        }
    }

    on_weekly_RUS = {
        effect = {
            if = {
                limit = { has_idea = idea_wheat_investments has_idea = idea_wheat_shortages }
                random = { chance = 40 country_event = kadets_opposition.1 }
            }

            if = {
                limit = { controls_state = 236 communism > 0.25 }
                news_event = trotsky_astrakhan.1
            }
        }
    }

    on_weekly_SER = {
        effect = {
            if = {
                limit = { has_completed_focus = XXX_extend    can_declare_war_on = TUR   has_global_flag = southeastern_war }
                country_event = { id = ottoman_opportunity.1  days = 2      random_days = 4 }
            }

            if = {
                limit = { has_country_flag = ottoman_war_winner     has_completed_focus = XXX_claims }
                country_event = albania_demilitarize.1
            }

            if = {
                limit = { has_global_flag = second_southeastern_war }
                if = {
                    limit = { 
                        war_length_with = { tag = TUR length < 3 }
                        every_owned_state = { is_controlled_by = ROOT }     controls_state = 44
                    }
                    country_event = albania_ending_serbia.1
                }
                if = {
                    limit = { war_length_with = { tag = TUR length > 4 }      TUR = { surrender_progress < 0.1 } }
                    country_event = albania_ending_turkey.1
                }

                if = {
                    limit = { not = { has_war_with = TUR } }
                    clr_global_flag = second_southeastern_war
                }
            }

            if = {
                limit = {
                    has_war_with = AUS      is_in_faction = no      AUS = { is_in_faction = no }
                    has_completed_focus = XXX_unity

                    controls_state = 103    controls_state = 104    controls_state = 109
                    not = { any_state = { is_owned_by = SER not = { is_controlled_by = SER } } }
                }
                country_event = savoie_ending_yugoslavia.1
            }
        }
    }

    on_weekly_SMS = {
        effect = {
            if = {
                limit = {
                    is_subject_of = TUR     TUR = { has_war_with = GRE }
                    not = { has_completed_focus = XXX_allegiance }
                }
                country_event = ottoman_samian_autonomy.1
            }
        }
    }

    on_weekly_TUR = {
        effect = {
            if = {
                limit = {
                    has_global_flag = southeastern_war
                    or = {
                        war_length_with = { tag = ROM months > 4 }
                        and = { war_length_with = { tag = ROM months > 3 } surrender_progress > 0.2 }
                    }
                }
                country_event = ottoman_young_turks.1 
            }

            if = {
                limit = { has_global_flag = southeastern_war      has_war = no }
                news_event = ottoman_armistice.2
            }

            # Act II
                if = {
                    limit = { has_idea = idea_oil_investments }

                    if = {
                        limit = { democratic > 0.15 }
                        country_event = francophiles_opposition.1
                    }
                    if = {
                        limit = { democratic > 0.25 }
                        country_event = francophiles_assyria.1
                    }
                    if = {
                        limit = { 
                            democratic > 0.25
                            monarchism < 0.5
                        }
                        country_event = francophiles_demonstrations.1
                    }
                    if = {
                        limit = { 
                            has_government = democratic 
                            FRA = { has_war_with = GER }
                        }
                        country_event = francophiles_entente.1
                    }
                }
        }
    }

    on_weekly_USA = {
        effect = {
            # Act I
                if = {
                    limit = { MEX = { has_civil_war = yes } }
                    news_event = { id = unions_agrarians.1 days = 2 random_days = 4 }
                }
                if = {
                    limit = { has_war_with = COL COL = { surrender_progress > 0.05 } }
                    country_event = mexico_armistice.1
                }

            # Act III
                if = {
                    limit = { POL = { has_idea = idea_uprising } }
                    country_event = kadets_airmen.1
                }
                if = {
                    limit = { POL = { has_idea = idea_uprising longest_war_length > 2 } }
                    country_event = kadets_washington.1
                }
        }
    }

    on_ace_promoted = {
        effect = {
            if = { 
                limit = { not = { has_global_flag = aces_used } }
                ROOT = { news_event = aces.1 }
            }
        }
    }

    on_new_term_election = {
        random_events = {
            100 = elections_belgium.1
            100 = elections_england.1
            100 = elections_france.1
            100 = elections_holland.1
            100 = elections_italy.1
        }
    }

# - generic event launcher
    on_monthly = {
        effect = {
            if = {
                limit = {
                    not = { has_idea = idea_anti_war_demonstrations }
                    or = {
                        and = { longest_war_length > 2      has_war_support < 0.25 }
                        and = { longest_war_length > 24     has_war_support < 0.4 }
                    }
                }
                random = { chance = 20      country_event = { id = anti_war.1    random_days = 28 } }
            }
            if = {
                limit = {
                    has_war = no
                    or = {
                        has_idea = partial_economic_mobilisation
                        has_idea = war_economy
                    }
                    not = {
                        or = {
                            has_idea = volunteer_only
                            has_idea = limited_conscription
                        }
                    }
                }
                random = { chance = 40      country_event = { id = demobilization.1    random_days = 28 } }
            }
            if = {
                limit = {
                    not = { has_idea = idea_widespread_hunger }
                    or = {
                        has_idea = partial_economic_mobilisation
                        has_idea = war_economy
                    }
                    or = {
                        and = { longest_war_length > 6      has_stability < 0.25 }
                        and = { longest_war_length > 12     has_stability < 0.4 }
                    }
                }
                random = { chance = 10      country_event = { id = hunger.1    random_days = 28 } }
            }
            if = {
                limit = {
                    has_civil_war = no
                    and = {
                        not = { has_idea = idea_uprising }
                        check_variable = { highest_party_popularity@exclude_ruling_party > 0.5 }
                    }
                }
                random = { chance = 40      country_event = { id = opposition.1    random_days = 28 } }
            }
            if = {
                limit = {
                    not = { has_idea = idea_strikes }
                    has_stability < 0.45
                }
                random = { chance = 40      country_event = { id = strikes.1    random_days = 28 } }
            }
        }
    }
}
