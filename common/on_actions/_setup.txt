on_actions = {
    on_startup = {
        effect = {

        # resource rights
            COL = {
                give_resource_rights = { receiver = USA state = 312 }
                give_resource_rights = { receiver = USA state = 317 }
            }

            MEX = {
                give_resource_rights = { receiver = USA state = 476 }
                give_resource_rights = { receiver = USA state = 485 }
            }

            QAJ = {
                give_resource_rights = { receiver = ENG state = 412 }
                give_resource_rights = { receiver = ENG state = 413 }
            }

        # events
            AUS = {
                news_event = { id = emperor_austria.1 months = 46 days = 27 }
            }

            ENG = {
                news_event = { id = titanic.1 months = 3 days = 14 }
            }

            FRA = {
                country_event = { id = caillaux.1 days = 12}
                news_event = { id = garros.2 months = 20 days = 3 }
            }

            GER = {
                news_event = { id = equestrian.1 days = 26 }
                news_event = { id = johannisthal.3 months = 21 days = 17 }
            }

            GRE = {
                news_event = { id = konstantinos.1 months = 14 days = 7 }
            }

            HOL = {
                news_event = { id = soesterberg.1 months = 7 days = 14 }
            }

            JAP = {
                news_event = { id = meiji.1 months = 6 days = 30 }
                country_event = { id = diet.1 months = 12 days = 7 }
            }

            MAN = {
                country_event = { id = tibet.1 months = 3 days = 4 }
            }

            MOR = {
                news_event = { id = atlas.1 months = 28 days = 6 }
            }

            RUS = {
                news_event = { id = muromets.1 months = 21 days = 17 }
            }

            TUR = {
                news_event = { id = aviators.1 months = 2 days = 20 }
            }

            USA = {
                news_event = { id = harvest.1 months = 68 }
                news_event = { id = wheat_shortage.1 months = 72 days = 14 }
            }
        }
    }

    on_daily = {
        effect = {
            # clean uprising tags
                if = {
                    limit = { 
                        has_idea = idea_uprising
                        or = { has_completed_focus = XXX_interim_government has_war = no }
                    }
                    country_event = other.3
                }       

            # warlord colors
                if = {
                    limit = { or = { is_chinese_warlord = yes original_tag = MAN } has_war_with = MAN }
                    set_cosmetic_tag = MAN_enemy_warlord
                }
                if = {
                    limit = { has_cosmetic_tag = MAN_enemy_warlord not = { has_war_with = MAN } }
                    drop_cosmetic_tag = yes
                }

            # autonomy loss
                if = {
                    limit = { is_subject = yes has_elections = yes }
                    set_politics = { ruling_party = ROOT elections_allowed = no }
                }
            
            #
        }
    }

    on_weekly = {
        effect = {
            if = {
                limit = { can_have_an_uprising = yes }
                every_possible_country = { limit = { can_start_uprising = yes } random = { chance = 5 trigger_uprising = yes } }
            }

            if = {
                limit = { 
                    has_civil_war = no
                    check_variable = { highest_party_popularity@exclude_ruling_party > 0.2 }

                    or = {
                        has_stability < 0.2
                        and = {
                            has_stability < 0.4
                            or = {
                                and = { surrender_progress > 0.1        has_war_support < 0.25 }
                                and = { surrender_progress > 0.2        has_war_support < 0.4 }
                            }
                        }
                    }   
                }
                random = { chance = 10 trigger_civil_war = yes }
            }

            if = {
                limit = { any_owned_state = { and = { days_since_last_strategic_bombing < 7 not = { has_global_flag = airships_used } } } }
                news_event = airships.1
            }

            if = {
                limit = { has_war = yes has_tech = gas_shells not = { has_global_flag = gas_used } }
                news_event = gas.1
            }

            # country specifics
                if = {
                    limit = { 
                        tag = RUS 
                        and = { communism > 0.25     has_stability < 0.4 } 
                    }
                    country_event = astrakhan.1
                }
        }
    }

    on_ace_promoted = {
        effect = {
            if = { 
                limit = { not = { has_global_flag = aces_used } }
                ROOT = { news_event = aces.1 }
            }
        }
    }

    on_new_term_election = {
        random_events = {
            100 = elections_england.1
            100 = elections_france.1
            100 = elections_holland.1
            100 = elections_italy.1
        }
    }

# - generic event launcher
    on_monthly = {
        effect = {
            if = {
                limit = {
                    not = { has_idea = idea_anti_war_demonstrations }
                    or = {
                        and = { longest_war_length > 2      has_war_support < 0.25 }
                        and = { longest_war_length > 24     has_war_support < 0.4 }
                    }
                }
                random = { chance = 20      country_event = { id = anti_war.1    random_days = 28 } }
            }
            if = {
                limit = {
                    has_war = no
                    or = {
                        has_idea = partial_economic_mobilisation
                        has_idea = war_economy
                    }
                    not = {
                        or = {
                            has_idea = volunteer_only
                            has_idea = limited_conscription
                        }
                    }
                }
                random = { chance = 40      country_event = { id = demobilization.1    random_days = 28 } }
            }
            if = {
                limit = {
                    not = { has_idea = idea_widespread_hunger }
                    or = {
                        has_idea = partial_economic_mobilisation
                        has_idea = war_economy
                    }
                    or = {
                        and = { longest_war_length > 6      has_stability < 0.25 }
                        and = { longest_war_length > 12     has_stability < 0.4 }
                    }
                }
                random = { chance = 10      country_event = { id = hunger.1    random_days = 28 } }
            }
            if = {
                limit = {
                    has_civil_war = no
                    and = {
                        not = { has_idea = idea_uprising }
                        check_variable = { highest_party_popularity@exclude_ruling_party > 0.5 }
                    }
                }
                random = { chance = 40      country_event = { id = opposition.1    random_days = 28 } }
            }
            if = {
                limit = {
                    not = { has_idea = idea_strikes }
                    has_stability < 0.45
                }
                random = { chance = 40      country_event = { id = strikes.1    random_days = 28 } }
            }
        }
    }
}
