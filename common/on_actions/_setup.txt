on_actions = {
    on_startup = {
        effect = {
            # template lock
                FER = {
                    country_lock_all_division_template = yes
                }

            # resource rights
                COL = {
                    give_resource_rights = { receiver = USA state = 312 }
                    give_resource_rights = { receiver = USA state = 317 }
                }

                MEX = {
                    give_resource_rights = { receiver = USA state = 476 }
                    give_resource_rights = { receiver = USA state = 485 }
                }

                QAJ = {
                    give_resource_rights = { receiver = ENG state = 412 }
                    give_resource_rights = { receiver = ENG state = 413 }
                }

            # events
                MAN = {
                    country_event = { id = manchukuo.1 months = 3 days = 2 }        # Tibetan Independence (2.04.1912)
                }
        }
    }

    on_daily = {
        effect = {
        # uprisings     
            if = {
                limit = { or = { has_stability < 0.2 has_civil_war = yes } }
                every_possible_country = {
                    limit = { 
                        and = {
                            any_core_state = { is_controlled_by = PREV.PREV }
                            is_austrian_subject = no is_chinese_warlord = no is_middle_eastern_turkish_subject = no 
                        }
                        exists = no has_capitulated = no not = { has_country_flag = country_defeated }
                    }
                    random = {
                        chance = 1
                        trigger_uprising = yes
                    }
                }
            }       

            # clean uprising tags
            if = {
                limit = { has_country_flag = during_uprising has_war = no }
                country_event = uprising.2
            }

        # liberation from overlords

            # clean dominion trees
            if = {
                limit = { has_country_flag = starts_as_dominion is_subject = no }
                mark_focus_tree_layout_dirty = yes
            }  

        # unifiers' ambitions
            if = {
                limit = {
                    can_choose_ambition = yes
                    not = { has_country_flag = had_ambition_event }
                }
                country_event = unification.1
            }

        # austria
            # set austrian tag if hungary exists
            if = {
                limit = { country_exists = HUN }
                AUS = { set_cosmetic_tag = AUS_alone }
            }

        # china
            # china tags
            if = {
                limit = { 
                    or = { original_tag = MAN is_chinese_warlord = yes } has_war_with = MAN 
                    not = { original_tag = FER }
                }
                set_cosmetic_tag = MAN_enemy_warlord
            }
            if = {
                limit = { not = { has_war_with = MAN } has_cosmetic_tag = MAN_enemy_warlord }
                drop_cosmetic_tag = yes
            }
            if = {
                limit = { has_cosmetic_tag = MAN_empire not = { has_government = monarchism } }
                drop_cosmetic_tag = yes
            }

        # india
            # cleanup of india tag
            if = {
                limit = { original_tag = RAJ is_subject = no }
                drop_cosmetic_tag = yes
            }
        }
    }

# specific countries
    on_daily_AUS = {
        effect = {
            # austrian disintegration
            if = {
                limit = {
                    or = { has_stability < 0.4 has_civil_war = yes }
                    has_idea = idea_decentralizing_tensions
                }

                # trigger random austrian subjects
                every_possible_country = {
                    limit = { 
                        is_austrian_subject = yes 
                        any_core_state = { is_controlled_by = AUS }
                        exists = no has_capitulated = no not = { has_country_flag = country_defeated }
                    }
                    random = {
                        chance = 3
                        trigger_subject = yes
                    }
                }
            }
        }
    }

    on_daily_MAN = {
        effect = {
            # chinese warlords setup
            if = {
                limit = {
                    or = { has_stability < 0.25 has_civil_war = yes }
                    has_idea = idea_warlordism_in_the_countryside
                }

                # trigger random warlord states
                every_possible_country = {
                    limit = { 
                        is_chinese_warlord = yes
                        any_core_state = { is_controlled_by = MAN }
                        exists = no has_capitulated = no not = { has_country_flag = country_defeated }
                    }
                    random = {
                        chance = 3
                        trigger_warlord = yes
                    }
                }
            }
        }
    }

    on_daily_TUR = {
        effect = {
            # turkish middle eastern subjects
            if = {
                limit = {
                    has_stability < 0.4
                    or = {  has_idea = idea_unrest_in_the_middle_east has_civil_war = yes }  
                }

                # trigger random turkish subjects
                every_possible_country = {
                    limit = { 
                        is_middle_eastern_turkish_subject = yes 
                        any_core_state = { is_controlled_by = TUR }
                        exists = no has_capitulated = no not = { has_country_flag = country_defeated }
                    }
                    random = {
                        chance = 1
                        trigger_subject = yes
                    }
                }
            }
        }
    }

# - elections
    on_new_term_election = {
        effect = {
        }
    }

# - generic event launcher
    on_monthly = {
        effect = {
            # unrest
            random = { chance = 40 news_event = { id = generic.1 random_days = 28 } }

            # ideology
            country_event = { id = generic.2 random_days = 28 }

            # gov't
            country_event = { id = generic.4 random_days = 28 }
            country_event = { id = generic.5 random_days = 28 }
            country_event = { id = generic.6 random_days = 28 }

            # war exhaustion
            random = { chance = 10 news_event = { id = generic.8 random_days = 28 } }

            # armistice
            news_event = { id = generic.11 random_days = 28 }
            country_event = { id = generic.12 random_days = 28 }
        }
    }
}
