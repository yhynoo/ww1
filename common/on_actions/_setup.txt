on_actions = {
    on_startup = {
        effect = {

        # resource rights
            COL = {
                give_resource_rights = { receiver = USA state = 312 }
                give_resource_rights = { receiver = USA state = 317 }
            }

            MEX = {
                give_resource_rights = { receiver = USA state = 476 }
                give_resource_rights = { receiver = USA state = 485 }
            }

            QAJ = {
                give_resource_rights = { receiver = ENG state = 412 }
                give_resource_rights = { receiver = ENG state = 413 }
            }

        # events
            
        }
    }

    on_weekly = {
        effect = {
            if = {
                limit = { has_country_flag = southeastern_europe_war has_war = no }
                clr_country_flag = southeastern_europe_war
                
                # event
                news_event = TUR_end.1
            }
        }
    }

    on_daily = {
        effect = {
        # uprisings  
            # clean uprising tags
            if = {
                limit = { has_country_flag = during_uprising has_war = no }
                country_event = uprising.4
            }

            # set up the uprising
            if = {
                limit = { can_fall_apart = yes }
                every_possible_country = {
                    limit = { 
                        is_chinese_state = no
                        is_german_state = no
                        can_trigger_uprising = yes
                    }
                    random = {
                        chance = 2
                        trigger_uprising = yes
                    }
                }
            }       

        # autonomy loss
            if = {
                limit = { is_subject = yes has_elections = yes }
                set_politics = { ruling_party = ROOT elections_allowed = no }
            }

        # tags 
            manage_client_state_tags = yes

        # austria
            if = {
                limit = { original_tag = AUS }
                if = {
                    limit = { country_exists = HUN }
                    drop_cosmetic_tag = yes
                }
            }

        # china
            if = {
                limit = { original_tag = MAN }
                if = {
                    limit = {
                        can_fall_apart = yes
                        has_idea = idea_warlordism_in_the_countryside
                    }

                    # trigger random warlord states
                    every_possible_country = {
                        limit = { 
                            is_chinese_state = yes
                            can_trigger_uprising = yes
                        }
                        random = {
                            chance = 3
                            trigger_warlord = yes
                        }
                    }
                }
            }
        }
    }

# - elections }
    on_new_term_election = {
        random_events = {
            100 = FRA_elections.1
            100 = HOL_elections.1
        }
    }

# - generic event launcher
    on_monthly = {
        effect = {
            # exhaustion
            random = { chance = 10 country_event = { id = exhaustion.1 random_days = 28 } }
            random = { chance = 10 country_event = { id = exhaustion.3 random_days = 28 } }

            # unrest
            random = { chance = 50 country_event = { id = unrest.1 random_days = 28 } }
        }
    }
}
