on_actions = {
    on_startup = {
        effect = {

        # resource rights
            COL = {
                give_resource_rights = { receiver = USA state = 312 }
                give_resource_rights = { receiver = USA state = 317 }
            }

            MEX = {
                give_resource_rights = { receiver = USA state = 476 }
                give_resource_rights = { receiver = USA state = 485 }
            }

            QAJ = {
                give_resource_rights = { receiver = ENG state = 412 }
                give_resource_rights = { receiver = ENG state = 413 }
            }

        # events
            AUS = {
                country_event = { id = aus_emperor.1 months = 58 days = 26 }
                country_event = { id = aus_streik.1 months = 85 days = 4 }
            }

            ENG = {
                news_event = { id = eng_halifax.1 months = 71 days = 6 }
            }

            FRA = {
                news_event = { id = fra_agadir.1 days = 12 }
                news_event = { id = fra_entente.1 months = 26 days = 6 }
                news_event = { id = fra_oil.1 months = 39 days = 10 }
            }

            GER = {
                news_event = { id = ger_occupation.1 months = 1 days = 10 }
                news_event = { id = ger_deal.1 months = 25 }

                news_event = { id = ger_wassmuss.1 months = 65 }
                news_event = { id = ger_harvest.1 months = 68 }
                news_event = { id = ger_harvest.2 months = 72 days = 14 }

                country_event = { id = ger_armistice.1 months = 86 days = 7 }
            }

            JAP = {
                news_event = { id = jap_emperor_dies.1 months = 6 days = 30 }
                country_event = { id = jap_address.1 months = 12 days = 6 }

                MAN = { country_event = { id = jap_twenty_one_demands.1 months = 41 days = 7 } }
            }

            MAN = {
                country_event = { id = man_transition.1 months = 2 days = 14 }
                country_event = { id = man_tibet.1 months = 3 days = 4 }
                news_event = { id = man_yuan.1 months = 53 days = 6 }
            }

            MEX = {
                news_event = { id = mex_agrarianists.1 months = 1 days = 6 }
            }

            MOR = {
                news_event = { id = mor_atlas.1 months = 29 days = 7 }
            }

            ROM = {
                every_country = {
                    limit = { or = { tag = GRE tag = SER } }
                    country_event = { id = rom_alliance.1 months = 10 }
                }

                # war
                news_event = { id = rom_war.1 months = 10 days = 14 }
            }

            SPA = {
                news_event = { id = spa_morocco.1 months = 16 }
                news_event = { id = spa_entente.1 months = 26 days = 12 }
            }

            USA = {
                country_event = { id = cub_uprising.1 months = 4 days = 21 }
                news_event = { id = usa_lafayette.1 months = 51 days = 7 }
                news_event = { id = usa_wilson.1 months = 80 }
            }
        }
    }

    # reactions - generic
        on_weekly = {
            effect = {
                if = {
                    limit = { any_owned_state = { and = { days_since_last_strategic_bombing < 7 not = { has_global_flag = airships_used } } } }
                    news_event = ger_airships.1
                }

                if = {
                    limit = { has_war = yes has_tech = gas_shells not = { has_global_flag = gas_used } }
                    news_event = ger_gas.1
                }
            }
        }

        on_ace_promoted = {
            effect = {
                if = { 
                    limit = { not = { has_global_flag = aces_used } }
                    ROOT = { news_event = ger_aces.1 }
                }
            }
        }

    # reactions - countries
        on_weekly_COL = {
            effect = {
                if = {
                    limit = {
                        and = { has_war_with = USA MEX = { is_subject_of = USA } }
                        or = {
                            not = { controls_state = 312 }  not = { controls_state = 317 }
                            surrender_progress > 0.05
                        }
                    }
                    USA = { country_event = col_armistice.1 }
                }
            }
        }
        on_weekly_ENG = {
            effect = {
                if = {
                    limit = {
                        has_global_flag = great_war
                        or = { SPA = { has_civil_war = yes } MOR = { controls_state = 290 } }
                    }
                    country_event = eng_danger.1
                }
            }
        }
        on_weekly_FRA = {
            effect = {
                if = {
                    limit = { MOR = { has_civil_war = yes } }
                    country_event = { id = fra_intervene.1 days = 3 }
                }
                if = {
                    limit = {
                        not = { has_global_flag = great_war }
                        has_war_with = GER
                    }
                    news_event = fra_war.1
                }
                if = {
                    limit = { 
                        has_global_flag = great_war
                        not = { has_war_with = GER }
                    }

                    clr_global_flag = great_war
                    news_event = fra_war.2
                }
            }
        }

        on_weekly_GER = {
            effect = {
                # flag
                if = {
                    limit = { has_government = monarchism }
                    if = {
                        limit = { has_war = yes not = { has_cosmetic_tag = GER_kriegsflagge } }
                        set_cosmetic_tag = GER_kriegsflagge
                    }
                    if = {
                        limit = { has_war = no has_cosmetic_tag = GER_kriegsflagge }
                        drop_cosmetic_tag = yes
                    }
                }

                if = {
                    limit = { war_length_with = { tag = TUR months = 2 } not = { has_country_flag = middle_east_influence } }
                    news_event = ger_middle_east.1
                }
            }
        }

        on_weekly_MEX = {
            effect = {
                if = {
                    limit = { has_civil_war = yes surrender_progress > 0.4 }
                    news_event = mex_americans.1
                }
            }
        }

        on_weekly_MOR = {
            effect = {
                if = { 
                    limit = { has_capitulated = yes not = { has_country_flag = end_of_morocco } }
                    news_event = mor_end.1
                }
            }
        }

        on_weekly_POL = {
            effect = {
                if = { 
                    limit = { 
                        owns_state = 80
                        and = { has_war = no is_in_faction = no }
                    } 
                    news_event = pol_entente.1
                }
                if = {
                    limit = {
                        not = { has_global_flag = great_war }
                        controls_state = 89
                    }
                    news_event = pol_lviv.1
                }
            }
        }

        on_weekly_ROM = {
            effect = {
                if = {
                    limit = { 
                        has_global_flag = southeastern_europe_war 
                        TUR = { has_war = no }
                    }

                    clr_global_flag = southeastern_europe_war
                    news_event = rom_war.2
                }
            }
        }

        on_weekly_SAU = {
            effect = {
                if = {
                    limit = { TUR = { has_government = democratic has_subject = SAU } }
                    TUR = { news_event = { id = sau_autonomy.1 days = 2 } }
                }
            }
        }

        on_weekly_SER = {
            effect = {
                if = {
                    limit = { has_war = no TUR = { has_war_with = AUS } }
                    country_event = ser_war.1
                }
            }
        }

        on_weekly_SPA = {
            effect = {
                if = {
                    limit = { casualties_k > 20 not = { has_country_flag = had_semana } }
                    news_event = spa_semana.1
                }

                if = {
                    limit = { war_length_with = { tag = GER months > 3 } = yes has_idea = idea_memories_of_semana_tragica communism > 0.3 }
                    country_event = spa_civil_war.1
                }
            }
        }

        on_weekly_THR = {
            effect = {
                if = {
                    limit = { TUR = { has_government = democratic has_subject = THR } }
                    ROM = { news_event = thr_guarantees.1 }
                }

                if = {
                    limit = {
                        TUR = { has_subject = THR }
                        is_guaranteed_by = ROM
                    }
                    TUR = { country_event = thr_autonomy.1 }
                }
            }
        }

        on_weekly_TUR = {
            effect = {
                # war in southeastern europe
                if = {
                    limit = { 
                        has_global_flag = southeastern_europe_war
                        and = {
                            ARM = { exists = yes }
                            or = { war_length_with = { tag = ROM months > 3 } surrender_progress > 0.15 }
                        }
                    }
                    country_event = tur_coup.1
                }

                # democratic turn
                if = {
                    limit = {
                        has_idea = idea_french_oil_investments
                        and = { has_government = monarchism democratic > 0.35 }
                    }
                    country_event = tur_demonstrations.1

                    # entente
                    news_event = { id = tur_entente.1 months = 2 }
                }
            }
        }

        on_weekly_USA = {
            effect = {
                if = {
                    limit = { 
                        and = {
                            has_war_with = CUB 
                            315 = { is_fully_controlled_by = CUB } 
                        }
                    }
                    news_event = cub_armistice.1
                }
            }
        }

    # generic events
        on_monthly = {
            effect = {
                # exhaustion
                random = { chance = 10 country_event = { id = exhaustion.1 random_days = 28 } }

                # unrest
                random = { chance = 50 country_event = { id = unrest.1 random_days = 28 } }
            }
        }
}
