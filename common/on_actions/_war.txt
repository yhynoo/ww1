on_actions = {
    on_war_relation_added = {
        effect = {
            if = {
                limit = { ROOT = { original_tag = FROM } }      # same tags: civil war.
                # news_event = { id = civil_war.4 hours = 6 }

                every_country = {
                    limit = { is_diplomatically_able = yes has_offensive_war_with = FROM not = { original_tag = FROM } }
                    country_event = { id = civil_war.1 hours = 6 }
                }
                
                FROM = { 
                    add_popularity = { ideology = FROM popularity = 0.15 }
                    add_popularity = { ideology = ROOT popularity = -0.25 } 
                }

                every_country = {
                    limit = { original_tag = ROOT has_civil_war = yes }
                    setup_civil_war = yes
                }
            }
        }
    }

    on_civil_war_end = {
        effect = {
            set_rule = { can_join_factions = yes }              # can join factions again.
            if = { limit = { not = { has_country_flag = after_civil_war } } set_country_flag = after_civil_war }
        }
    }

    on_annex = {
        effect = {
            if = {
                limit = {
                    or = {
                        is_chinese_state = yes
                        has_country_flag = during_uprising
                    }
                }
                FROM = { set_country_flag = country_defeated }
            }
        }
    }
}
