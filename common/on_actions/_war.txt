on_actions = {
    on_war_relation_added = {
        effect = {
            if = {
                limit = { ROOT = { original_tag = FROM } }      # same tags: civil war.
                news_event = { id = civil_war.4 hours = 6 }

                every_country = {
                    limit = { is_diplomatically_able = yes has_offensive_war_with = FROM }
                    country_event = { id = civil_war.1 hours = 6 }
                }
                
                FROM = { 
                    add_popularity = { ideology = FROM popularity = 0.15 }
                    add_popularity = { ideology = ROOT popularity = -0.25 } 
                }

                every_country = {
                    limit = { original_tag = ROOT has_civil_war = yes }
                    setup_civil_war = yes
                }
            }

            if = {
                limit = { any_country = { is_subject_of = ROOT has_completed_focus = XXX_autonomy } }
                every_country = {
                    limit = { is_subject_of = ROOT has_completed_focus = XXX_autonomy }
                    country_event = { id = dominion.7 days = 2 }
                }
            }

            if = {
                limit = { 
                    original_tag = GER 
                    FROM = { original_tag = FRA }
                }
                news_event = { id = europe_war.1 hours = 6 }
            }
        }
    }

    on_civil_war_end = {
        effect = {
            set_rule = { can_join_factions = yes }              # can join factions again.
            if = { limit = { not = { has_country_flag = after_civil_war } } set_country_flag = after_civil_war }
        }
    }

    on_peaceconference_ended = {
        effect = {
            if = {
                limit = {
                    # if we have the great war idea, but the following conditions are not met - the great war has ended.
                    has_idea = idea_great_war
                    not = {
                        and = {
                            or = { is_major = yes any_country = { has_war_together_with = ROOT is_major = yes } }
                            any_enemy_country = is_major
                        }
                    }
                }
                add_timed_idea = { idea = idea_war_winner months = 12 }

                # the great war has ended.
                every_country = {
                    limit = { has_idea = idea_great_war }
                    remove_ideas = idea_great_war
                }
            }

            FROM = { 
                if = {
                    limit = {
                        # if we have the great war idea, but the following conditions are not met - the great war has ended.
                        has_idea = idea_great_war
                        not = {
                            and = {
                                or = { is_major = yes any_country = { has_war_together_with = ROOT is_major = yes } }
                                any_enemy_country = is_major
                            }
                        }
                    }
                    add_timed_idea = { idea = idea_war_loser months = 18 } 
                }
            }
        }
    }

    on_annex = {
        effect = {
            if = {
                limit = {
                    or = {
                        is_chinese_state = yes
                        has_country_flag = during_uprising
                    }
                }
                FROM = { set_country_flag = country_defeated }
            }
        }
    }
}
